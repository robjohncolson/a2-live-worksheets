function advanceDailyQueue() {
  if (!state.dailyDose || !state.activeProbe) return null;
  const currentId = state.activeProbe.questionId;
  const next = advanceDailyQueueV5(state, currentId);
  if (!next) {
    state.activeProbe = null;
    saveSoon();
