```js
      function renderQueuePane() {
        if (!ui.rail) return;
        ui.rail.innerHTML = '';
        const today = new Date().toISOString().slice(0, 10);
        if (!state.dailyDose || state.dailyDose.date !== today) {
          state.dailyDose = pickDailyQueue(state, today, AP_EXAM_DATE);
          saveSoon();
        }
        const dose = state.dailyDose;
        if (!dose || !Array.isArray(dose.queue) || dose.queue.length === 0) {
          ui.rail.appendChild(make('h3', '', 'Today'));
          ui.rail.appendChild(make('p', 'sg-empty', 'No queue items are available yet.'));
          return;
        }
        const header = make('div', 'sg-queue-header');
        header.appendChild(make('h3', '', `Today: ${dose.queue.length} question${dose.queue.length === 1 ? '' : 's'}`));
        const days = daysLeft(today, AP_EXAM_DATE);
        header.appendChild(make('div', 'sg-queue-sub', `${days} day${days === 1 ? '' : 's'} until the AP exam.`));
        ui.rail.appendChild(header);

        const list = make('ol', 'sg-queue-list');
        dose.queue.forEach((entry, index) => {
          const item = make('li', 'sg-queue-item');
          const done = Array.isArray(dose.completed) && dose.completed.includes(entry.questionId);
          const isActive = state.activeProbe && state.activeProbe.questionId === entry.questionId;
          if (done) item.classList.add('is-done');
          if (isActive) item.classList.add('is-active');
          const button = make('button', 'sg-queue-btn');
          button.type = 'button';
          button.dataset.formulaId = entry.formulaId;
          button.dataset.questionId = entry.questionId;
          button.dataset.unit = String(entry.unit);
          button.innerHTML = `<span class="sg-queue-index">${index + 1}.</span> <span class="sg-queue-label">${esc(formulaName(entry.formulaId))}</span><span class="sg-queue-meta">Unit ${entry.unit} · L${entry.lesson}</span>`;
          button.addEventListener('click', () => {
            setActiveProbe(entry.unit, entry.questionId, 'daily-queue', entry.formulaId);
            renderActiveProbe();
            renderRemediation();
            renderQueuePane();
          });
          item.appendChild(button);
          list.appendChild(item);
        });
        ui.rail.appendChild(list);
        ui.rail.appendChild(renderSeeAllDisclosure());
      }

      function renderSeeAllDisclosure() {
        const wrapper = make('details', 'sg-see-all');
        wrapper.appendChild(make('summary', '', 'See all formulas'));
        const body = make('div', 'sg-see-all-body');
        const touchedCore = touchedFormulaCount(state, 'core');
        const touchedRegular = touchedFormulaCount(state, 'regular');
        const touchedPower = touchedFormulaCount(state, 'power');
        const touchedSupport = touchedFormulaCount(state, 'support');
        body.appendChild(make('p', 'sg-see-all-summary', `Touched so far: ${touchedCore} core · ${touchedRegular} regular · ${touchedPower} power · ${touchedSupport} support.`));
        const allFormulas = Object.keys(FORMULA_UNIT_MAP).sort();
        const grid = make('div', 'sg-see-all-grid');
        allFormulas.forEach(formulaId => {
          const mastery = formulaMastery(formulaId, state);
          const pct = Math.round(mastery * 100);
          const cell = make('div', 'sg-see-all-cell');
          cell.appendChild(make('div', 'sg-see-all-name', formulaName(formulaId)));
          cell.appendChild(make('div', 'sg-see-all-mastery', `${pct}%`));
          grid.appendChild(cell);
        });
        body.appendChild(grid);
        wrapper.appendChild(body);
        return wrapper;
      }

      function advanceDailyQueue() {
        if (!state.dailyDose || !Array.isArray(state.dailyDose.queue)) return null;
        const current = state.activeProbe && state.activeProbe.questionId;
        if (current && !state.dailyDose.completed.includes(current)) {
          state.dailyDose.completed.push(current);
        }
        const remaining = state.dailyDose.queue.find(entry => !state.dailyDose.completed.includes(entry.questionId));
        if (!remaining) {
          state.activeProbe = null;
          saveSoon();
          return null;
        }
        setActiveProbe(remaining.unit, remaining.questionId, 'daily-queue', remaining.formulaId);
        return remaining;
      }
```
