### Grading fetch pattern (copy this structure verbatim into your JS)
```js
async function gradeGate(unitNumber, questionId, answer) {
  const fullPrompt = findGateQuestionText(questionId); // look up from EMBEDDED_CURRICULUM
  const prompt = window.buildReflectionPromptSG(unitNumber, fullPrompt, answer);

  const response = await fetch(`${window.RAILWAY_SERVER_URL}/api/ai/grade`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({
      scenario: {
        topic: 'Algebra 2 Diagnostic Study Guide',
        questionId: questionId,
        lessonContext: window.LESSON_CONTEXT_SG,
      },
      answers: { answer: answer },
      prompt: prompt,
    }),
  });

  if (!response.ok) throw new Error('Grading request failed');
  const result = await response.json();

  // Enrichment pass: P with no missing elements → upgrade to E
  if (result.score === 'P' && result.missing && result.missing.length === 0) {
    result.score = 'E';
    result.feedback = 'Nice work — you covered the key ideas.';
  }
  return result;
}
```

### Railway answer submission (for class-wide tracking, optional and best-effort)
```js
async function submitGateAnswer(unitNumber, answer, result) {
  if (!window.railwayClient || !window.railwayClient.submitAnswer) return;
  try {
    await window.railwayClient.submitAnswer(
      `SG-U${unitNumber}-GATE`,
      answer,
      { score: result.score }
    );
  } catch (err) {
    console.warn('Railway submission failed (non-fatal):', err);
  }
}
```

## Probe selection (runs in browser when a drill-down opens)

```js
function buildProbesForUnit(unitNumber) {
  const unitPrefix = 'U' + unitNumber + '-L';
  const mcByLesson = {};
  for (const q of EMBEDDED_CURRICULUM) {
    if (q.type !== 'multiple-choice') continue;
    if (!q.id.startsWith(unitPrefix)) continue;
    const m = q.id.match(/^U\d+-L(\d+)-Q\d+/);
    if (!m) continue;
    const lesson = parseInt(m[1], 10);
    (mcByLesson[lesson] ||= []).push(q);
  }
  const lessons = Object.keys(mcByLesson).map(Number).sort((a, b) => a - b);
  const probes = [];
  for (const lesson of lessons.slice(0, 6)) {
    const first = mcByLesson[lesson].slice().sort((a, b) => a.id.localeCompare(b.id))[0];
    probes.push({ lesson: lesson, question: first });
  }
  return probes;
}
```
